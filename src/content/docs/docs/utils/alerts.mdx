## API

### `Alert.success(title, text?, timer?)`

- Muestra un modal de éxito.
- Por defecto se cierra solo.

```ts
await Alert.success('Guardado', 'Cambios aplicados correctamente');
await Alert.success('Listo', 'Operación completada', 1200);
```

### `Alert.error(title, text?)`

- Muestra un modal de error con botón **Entendido**.

```ts
await Alert.error('Error', 'No se pudo procesar la solicitud');
```

### `Alert.confirm(title, text, confirmText?, cancelText?)`

- Devuelve `true` si el usuario confirma.

```ts
const ok = await Alert.confirm(
  'Eliminar registro',
  'Esta acción no se puede deshacer',
  'Sí, eliminar',
  'Cancelar'
);

if (ok) {
  // ...
}
```

## Ejemplos reales (como Login/Register)

<Aside type="note">
  Los ejemplos de abajo siguen el mismo patrón que usamos en el flujo real: mostrar `Alert.error(...)` en el `catch` y `Alert.success(...)` antes de redirigir.
</Aside>

### Ejemplo: Login

```tsx
import { useState, type SyntheticEvent } from 'react';
import { Alert } from '@/utils/alerts';
import { loginWithCredentials, persistLogin, dashboardPathForUser } from '@/lib/services/auth/auth.service';

export function ExampleLogin() {
  const [ci, setCi] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);

  const onSubmit = async (e: SyntheticEvent<HTMLFormElement>) => {
    e.preventDefault();
    setLoading(true);

    try {
      const data = await loginWithCredentials({ ci: ci.trim(), password });
      persistLogin(data);

      await Alert.success('Inicio de sesión exitoso', `Bienvenido/a, ${data.user.name}`, 900);
      window.location.href = dashboardPathForUser(data.user, data.user.role?.code);
    } catch (err) {
      await Alert.error(
        'No se pudo iniciar sesión',
        err instanceof Error ? err.message : 'Verifica tus credenciales e inténtalo nuevamente.'
      );
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={onSubmit}>
      <input value={ci} onChange={(e) => setCi(e.target.value)} disabled={loading} />
      <input value={password} onChange={(e) => setPassword(e.target.value)} disabled={loading} />
      <button type="submit" disabled={loading}>Entrar</button>
    </form>
  );
}
```

### Ejemplo: Register

```tsx
import { useState, type SyntheticEvent } from 'react';
import { Alert } from '@/utils/alerts';
import { registerUser } from '@/lib/services/auth/auth.service';

export function ExampleRegister() {
  const [ci, setCi] = useState('');
  const [name, setName] = useState('');
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  const [loading, setLoading] = useState(false);

  const onSubmit = async (e: SyntheticEvent<HTMLFormElement>) => {
    e.preventDefault();

    if (password !== confirmPassword) {
      await Alert.error('Las contraseñas no coinciden', 'Por favor verifica e inténtalo de nuevo.');
      return;
    }

    setLoading(true);

    try {
      await registerUser({ ci: ci.trim(), name: name.trim(), password });
      await Alert.success('Cuenta creada', 'Ya puedes iniciar sesión.', 1400);
      window.location.href = '/login';
    } catch (err) {
      await Alert.error(
        'No se pudo crear el usuario',
        err instanceof Error ? err.message : 'Inténtalo nuevamente más tarde.'
      );
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={onSubmit}>
      <input value={ci} onChange={(e) => setCi(e.target.value)} disabled={loading} />
      <input value={name} onChange={(e) => setName(e.target.value)} disabled={loading} />
      <input value={password} onChange={(e) => setPassword(e.target.value)} disabled={loading} />
      <input value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} disabled={loading} />
      <button type="submit" disabled={loading}>Crear cuenta</button>
    </form>
  );
}
```
